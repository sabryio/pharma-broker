## Phase 1: Database Health & Connectivity

### Objective

Verify database accessibility, confirm all 17 expected tables exist, and get baseline row counts.

### Expected Deliverables

- ✅ Successful database connection
- ✅ All 17 tables present
- ✅ Row count baseline for trend analysis

### Script: `01_health_check.py`

```python
#!/usr/bin/env python3
"""
Phase 1: Database Health Check
==============================
Verifies database connectivity and table existence.

Expected Output:
- Connection status
- List of all tables with row counts
- Missing table warnings
"""

import pandas as pd
from sqlalchemy import create_engine, text, inspect
import os
import sys

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')

EXPECTED_TABLES = {
    'raw_messages': 'Incoming WhatsApp messages before AI processing',
    'offers': 'Medication supply offers extracted from messages',
    'requests': 'Medication demand requests extracted from messages',
    'matches': 'Offer-Request pairings with match scores',
    'groups': 'Monitored WhatsApp groups',
    'medication_mappings': 'Arabic→English medication name translations',
    'review_queue': 'Messages requiring manual review',
    'match_feedback': 'Operator feedback on match decisions',
    'unmapped_medications': 'Medications not found in mapping database',
    'feedback_records': 'Structured feedback for ML learning',
    'weight_history': 'Matching algorithm weight adjustments',
    'config': 'Application configuration key-value pairs',
    'audit_logs': 'System audit trail',
    'demand_leaderboard': 'Medication demand statistics',
    'match_queue': 'Pending match processing jobs',
    'failed_messages': 'Messages that failed AI processing',
    'bot_users': 'WhatsApp bot user preferences',
}

def main():
    print("=" * 70)
    print("PHASE 1: DATABASE HEALTH CHECK")
    print("=" * 70)

    # Step 1: Test Connection
    print("\n📡 Step 1: Testing Database Connection...")
    print("-" * 50)

    try:
        engine = create_engine(DATABASE_URL, pool_pre_ping=True)
        with engine.connect() as conn:
            result = conn.execute(text("SELECT version()"))
            version = result.scalar()
        print(f"✅ Connected successfully!")
        print(f"   PostgreSQL: {version[:50]}...")
    except Exception as e:
        print(f"❌ Connection FAILED: {e}")
        sys.exit(1)

    # Step 2: Check Tables
    print("\n📋 Step 2: Checking Expected Tables...")
    print("-" * 50)

    inspector = inspect(engine)
    existing_tables = set(inspector.get_table_names())

    missing = []
    found = []

    for table, desc in EXPECTED_TABLES.items():
        if table in existing_tables:
            found.append(table)
            print(f"  ✅ {table}")
        else:
            missing.append(table)
            print(f"  ❌ {table} - MISSING!")

    # Step 3: Row Counts
    print("\n📊 Step 3: Table Row Counts...")
    print("-" * 50)

    counts = []
    for table in sorted(found):
        try:
            df = pd.read_sql(f"SELECT COUNT(*) as cnt FROM {table}", engine)
            count = df['cnt'].iloc[0]
            counts.append({'table': table, 'rows': count, 'status': '✅'})
        except Exception as e:
            counts.append({'table': table, 'rows': 'ERROR', 'status': '❌'})

    # Sort by row count descending
    counts_df = pd.DataFrame(counts)
    counts_df = counts_df.sort_values('rows', ascending=False, key=lambda x: pd.to_numeric(x, errors='coerce'))

    print(f"  {'Table':<25} {'Rows':>12} {'Status'}")
    print(f"  {'-'*25} {'-'*12} {'-'*6}")
    for _, row in counts_df.iterrows():
        print(f"  {row['table']:<25} {str(row['rows']):>12} {row['status']}")

    # Summary
    print("\n" + "=" * 70)
    print("SUMMARY")
    print("=" * 70)
    total_rows = counts_df[counts_df['rows'] != 'ERROR']['rows'].astype(int).sum()
    print(f"  Tables Found:   {len(found)}/{len(EXPECTED_TABLES)}")
    print(f"  Tables Missing: {len(missing)}")
    print(f"  Total Rows:     {total_rows:,}")

    if missing:
        print(f"\n  ⚠️  Missing tables: {', '.join(missing)}")
    else:
        print(f"\n  ✅ All expected tables present!")

if __name__ == "__main__":
    main()
```

### Script: `02_completeness_analysis.py`

```python
#!/usr/bin/env python3
"""
Phase 2: Data Completeness Analysis
====================================
Identifies NULL values and assesses data population rates.

Expected Output:
- NULL percentage per column
- Critical column violations
- Recommendations for data quality improvement
"""

import pandas as pd
from sqlalchemy import create_engine
import os

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')
engine = create_engine(DATABASE_URL)

# Columns that must NOT be NULL
CRITICAL_COLUMNS = {
    'raw_messages': ['id', 'group_jid', 'content', 'sender_phone', 'timestamp'],
    'offers': ['id', 'medication', 'medication_raw', 'source_phone', 'raw_message', 'status'],
    'requests': ['id', 'medication', 'medication_raw', 'source_phone', 'raw_message', 'status'],
    'matches': ['id', 'offer_id', 'request_id', 'score', 'status'],
    'groups': ['jid', 'name'],
    'medication_mappings': ['id', 'arabic_name', 'english_name'],
}

# Columns expected to be NULL sometimes
EXPECTED_NULLABLE = {
    'raw_messages': ['processed_at', 'error', 'reply_to_id', 'reply_to_content', 'sender_name'],
    'offers': ['price', 'expiry_date', 'batch_number', 'notes', 'unit'],
    'requests': ['max_price', 'notes', 'unit'],
    'matches': ['confirmed_at', 'notes', 'reasoning'],
}

def analyze_table(table_name: str) -> pd.DataFrame:
    """Analyze NULL values for each column in a table."""
    try:
        df = pd.read_sql_table(table_name, engine)
    except Exception as e:
        print(f"  ❌ Error reading {table_name}: {e}")
        return pd.DataFrame()

    if len(df) == 0:
        print(f"  ⚠️  {table_name} is empty")
        return pd.DataFrame()

    results = []
    for col in df.columns:
        null_count = df[col].isna().sum()
        total = len(df)
        null_pct = round(null_count / total * 100, 2)

        is_critical = col in CRITICAL_COLUMNS.get(table_name, [])
        is_expected_null = col in EXPECTED_NULLABLE.get(table_name, [])

        severity = 'OK'
        if is_critical and null_count > 0:
            severity = '🚨 CRITICAL'
        elif not is_expected_null and null_pct > 50:
            severity = '⚠️  HIGH'
        elif not is_expected_null and null_pct > 20:
            severity = 'ℹ️  MODERATE'

        results.append({
            'column': col,
            'null_count': null_count,
            'total': total,
            'null_pct': null_pct,
            'is_critical': is_critical,
            'severity': severity
        })

    return pd.DataFrame(results)

def main():
    print("=" * 70)
    print("PHASE 2: DATA COMPLETENESS ANALYSIS")
    print("=" * 70)

    tables_to_check = ['raw_messages', 'offers', 'requests', 'matches', 'groups', 'medication_mappings']
    all_issues = []

    for table in tables_to_check:
        print(f"\n📋 Analyzing: {table}")
        print("-" * 60)

        analysis = analyze_table(table)
        if analysis.empty:
            continue

        # Show columns with issues
        issues = analysis[analysis['severity'] != 'OK'].sort_values('null_pct', ascending=False)

        if len(issues) == 0:
            print("  ✅ All columns have acceptable NULL rates")
        else:
            for _, row in issues.iterrows():
                print(f"  {row['severity']:12} {row['column']:<20} {row['null_count']:>6}/{row['total']} ({row['null_pct']}%)")
                if 'CRITICAL' in row['severity']:
                    all_issues.append({
                        'table': table,
                        'column': row['column'],
                        'null_count': row['null_count'],
                        'null_pct': row['null_pct']
                    })

        # Show summary
        total_cols = len(analysis)
        clean_cols = len(analysis[analysis['null_count'] == 0])
        print(f"\n  Summary: {clean_cols}/{total_cols} columns fully populated")

    # Critical Issues Summary
    print("\n" + "=" * 70)
    print("CRITICAL ISSUES REQUIRING ATTENTION")
    print("=" * 70)

    if all_issues:
        for issue in all_issues:
            print(f"  🚨 {issue['table']}.{issue['column']}: {issue['null_count']} NULLs ({issue['null_pct']}%)")
        print(f"\n  ⚠️  {len(all_issues)} critical column(s) have NULL values!")
    else:
        print("  ✅ No critical NULL issues found!")

if __name__ == "__main__":
    main()
```

### Interpreting Results

| Severity    | Meaning                   | Action Required         |
| ----------- | ------------------------- | ----------------------- |
| 🚨 CRITICAL | Required column has NULLs | Immediate investigation |
| ⚠️ HIGH     | >50% NULL unexpectedly    | Check data source       |
| ℹ️ MODERATE | 20-50% NULL unexpectedly  | Monitor trend           |
| OK          | Expected or acceptable    | No action               |

## Phase 3: Data Accuracy Verification

### Objective

Validate that data values fall within expected ranges and follow correct formats.

### Expected Deliverables

- ✅ Status enum validation
- ✅ Score range validation (0.0-1.0)
- ✅ Phone number format validation
- ✅ Quantity/price range checks

### Script: `03_accuracy_verification.py`

```python
#!/usr/bin/env python3
"""
Phase 3: Data Accuracy Verification
====================================
Validates data values against expected ranges and formats.
"""

import pandas as pd
from sqlalchemy import create_engine
import re
import os

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')
engine = create_engine(DATABASE_URL)

def verify_offers():
    """Verify offers table data accuracy."""
    print("\n📊 OFFERS TABLE VERIFICATION")
    print("-" * 50)

    df = pd.read_sql_table('offers', engine)
    if df.empty:
        print("  ⚠️  No offers found")
        return

    issues = []

    # 1. Medication names not empty
    empty_meds = (df['medication'].str.strip() == '').sum()
    status = '❌' if empty_meds > 0 else '✅'
    print(f"  {status} Empty medication names: {empty_meds}")
    if empty_meds > 0:
        issues.append(f"Empty medications: {empty_meds}")

    # 2. Quantity is non-negative
    neg_qty = (df['quantity'] < 0).sum()
    status = '❌' if neg_qty > 0 else '✅'
    print(f"  {status} Negative quantities: {neg_qty}")

    # 3. Status values are valid
    valid_statuses = ['ACTIVE', 'MATCHED', 'EXPIRED', 'ARCHIVED']
    invalid_status = ~df['status'].isin(valid_statuses)
    status = '❌' if invalid_status.sum() > 0 else '✅'
    print(f"  {status} Invalid status values: {invalid_status.sum()}")
    if invalid_status.sum() > 0:
        print(f"      Found: {df[invalid_status]['status'].unique().tolist()}")

    # 4. Price range (if present)
    if 'price' in df.columns:
        prices = df['price'].dropna()
        if len(prices) > 0:
            neg_prices = (prices < 0).sum()
            very_high = (prices > 100000).sum()  # EGP upper bound
            print(f"  {'❌' if neg_prices else '✅'} Negative prices: {neg_prices}")
            print(f"  {'⚠️' if very_high else '✅'} Very high prices (>100k): {very_high}")

    # 5. Statistics
    print(f"\n  📈 Statistics:")
    print(f"      Total offers: {len(df):,}")
    print(f"      With price: {df['price'].notna().sum():,}")
    print(f"      With quantity: {(df['quantity'] > 0).sum():,}")
    print(f"      Status distribution:")
    for status, count in df['status'].value_counts().items():
        print(f"          {status}: {count:,}")

def verify_requests():
    """Verify requests table data accuracy."""
    print("\n📊 REQUESTS TABLE VERIFICATION")
    print("-" * 50)

    df = pd.read_sql_table('requests', engine)
    if df.empty:
        print("  ⚠️  No requests found")
        return

    # Similar checks as offers
    empty_meds = (df['medication'].str.strip() == '').sum()
    print(f"  {'❌' if empty_meds else '✅'} Empty medication names: {empty_meds}")

    valid_statuses = ['ACTIVE', 'MATCHED', 'EXPIRED', 'ARCHIVED']
    invalid_status = ~df['status'].isin(valid_statuses)
    print(f"  {'❌' if invalid_status.sum() else '✅'} Invalid status values: {invalid_status.sum()}")

    # Urgent flag distribution
    urgent_count = df['urgent'].sum()
    urgent_pct = urgent_count / len(df) * 100
    print(f"  ℹ️  Urgent requests: {urgent_count:,} ({urgent_pct:.1f}%)")

    print(f"\n  📈 Statistics:")
    print(f"      Total requests: {len(df):,}")
    print(f"      Status distribution:")
    for status, count in df['status'].value_counts().items():
        print(f"          {status}: {count:,}")

def verify_matches():
    """Verify matches table data accuracy."""
    print("\n📊 MATCHES TABLE VERIFICATION")
    print("-" * 50)

    df = pd.read_sql_table('matches', engine)
    if df.empty:
        print("  ⚠️  No matches found")
        return

    # 1. Score range [0, 1]
    out_of_range = (df['score'] < 0) | (df['score'] > 1)
    print(f"  {'❌' if out_of_range.sum() else '✅'} Scores outside [0,1]: {out_of_range.sum()}")

    # 2. Score statistics
    print(f"  ℹ️  Score stats: min={df['score'].min():.3f}, max={df['score'].max():.3f}, mean={df['score'].mean():.3f}, median={df['score'].median():.3f}")

    # 3. Status validation
    valid_statuses = ['PENDING', 'CONFIRMED', 'REJECTED']

    print(f"      Total matches: {len(df):,}")
    print(f"      Status distribution:")
    for status, count in df['status'].value_counts().items():
        pct = count / len(df) * 100

    print("=" * 70)
```

## Phase 4: Relationship Integrity

### Objective

Verify foreign key relationships and detect orphaned records.

### Expected Deliverables

- ✅ All offer→raw_message links valid
- ✅ All match→offer and match→request links valid
- ✅ No duplicate match pairs

### Script: `04_relationship_integrity.py`

```python
#!/usr/bin/env python3
"""
Phase 4: Relationship Integrity Check
======================================
Verifies foreign key relationships and detects orphaned records.
"""

import pandas as pd
from sqlalchemy import create_engine
import os

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')
engine = create_engine(DATABASE_URL)

def check_relationship(name, query, desc):
    """Run a relationship check query."""
    result = pd.read_sql(query, engine)
    count = result.iloc[0, 0]
    status = '❌' if count > 0 else '✅'
    print(f"  {status} {desc}: {count}")
    return count

def main():
    print("=" * 70)
    print("PHASE 4: RELATIONSHIP INTEGRITY")
    print("=" * 70)

    issues = 0

    # Offers → Raw Messages
    print("\n📋 Offers → Raw Messages")
    print("-" * 50)
    issues += check_relationship(
        "orphan_offers",
        """
        SELECT COUNT(*) FROM offers o
        LEFT JOIN raw_messages rm ON o.raw_message_id = rm.id
        WHERE o.raw_message_id IS NOT NULL AND rm.id IS NULL
        """,
        "Offers with invalid raw_message_id"
    )

    # Requests → Raw Messages
    print("\n📋 Requests → Raw Messages")
    print("-" * 50)
    issues += check_relationship(
        "orphan_requests",
        """
        SELECT COUNT(*) FROM requests r
        LEFT JOIN raw_messages rm ON r.raw_message_id = rm.id
        WHERE r.raw_message_id IS NOT NULL AND rm.id IS NULL
        """,
        "Requests with invalid raw_message_id"
    )

    # Matches → Offers
    print("\n📋 Matches → Offers")
    print("-" * 50)
    issues += check_relationship(
        "orphan_match_offers",
        """
        SELECT COUNT(*) FROM matches m
        LEFT JOIN offers o ON m.offer_id = o.id
        WHERE o.id IS NULL
        """,
        "Matches with invalid offer_id"
    )

    # Matches → Requests
    print("\n📋 Matches → Requests")
    print("-" * 50)
    issues += check_relationship(
        "orphan_match_requests",
        """
        SELECT COUNT(*) FROM matches m
        LEFT JOIN requests r ON m.request_id = r.id
        WHERE r.id IS NULL
        """,
        "Matches with invalid request_id"
    )

    # Duplicate Matches
    print("\n📋 Duplicate Match Pairs")
    print("-" * 50)
    dupes = pd.read_sql("""
        SELECT offer_id, request_id, COUNT(*) as cnt
        FROM matches
        GROUP BY offer_id, request_id
        HAVING COUNT(*) > 1
    """, engine)
    status = '⚠️' if len(dupes) > 0 else '✅'
    print(f"  {status} Duplicate offer-request pairs: {len(dupes)}")
    issues += len(dupes)

    # Summary
    print("\n" + "=" * 70)
    if issues > 0:
        print(f"⚠️  {issues} relationship issue(s) found!")
    else:
        print("✅ All relationship integrity checks passed!")
    print("=" * 70)

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""
Phase 5: Business Logic Validation
===================================
Verifies data patterns match expected business logic.
"""

import pandas as pd
from sqlalchemy import create_engine
from datetime import datetime, timedelta
import os

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')
engine = create_engine(DATABASE_URL)

def analyze_conversion_rates():
    """Analyze message to offer/request conversion rates."""
    print("\n📊 MESSAGE CONVERSION ANALYSIS")
    print("-" * 60)

    raw_count = pd.read_sql("SELECT COUNT(*) FROM raw_messages", engine).iloc[0, 0]
    processed = pd.read_sql("SELECT COUNT(*) FROM raw_messages WHERE processed_at IS NOT NULL", engine).iloc[0, 0]

    offer_msgs = pd.read_sql("""
        SELECT COUNT(DISTINCT raw_message_id) FROM offers WHERE raw_message_id IS NOT NULL
    """, engine).iloc[0, 0]

    request_msgs = pd.read_sql("""
        SELECT COUNT(DISTINCT raw_message_id) FROM requests WHERE raw_message_id IS NOT NULL
    """, engine).iloc[0, 0]

    review_count = pd.read_sql("SELECT COUNT(*) FROM review_queue", engine).iloc[0, 0]
    failed_count = pd.read_sql("SELECT COUNT(*) FROM failed_messages", engine).iloc[0, 0]

    print(f"  Total raw messages:      {raw_count:,}")
    print(f"  Processed:               {processed:,} ({processed/raw_count*100:.1f}%)" if raw_count else "  N/A")
    print(f"  → Produced offers:       {offer_msgs:,} ({offer_msgs/raw_count*100:.1f}%)" if raw_count else "")
    print(f"  → Produced requests:     {request_msgs:,} ({request_msgs/raw_count*100:.1f}%)" if raw_count else "")
    print(f"  → In review queue:       {review_count:,} ({review_count/raw_count*100:.1f}%)" if raw_count else "")
    print(f"  → Failed:                {failed_count:,}")

    # Expected conversions analysis
    print("\n  📈 Conversion Health:")
    if raw_count > 0:
        no_output = raw_count - offer_msgs - request_msgs - review_count - failed_count
        if no_output > raw_count * 0.3:
            print(f"  ⚠️  High no-output rate: {no_output:,} messages ({no_output/raw_count*100:.1f}%)")
        else:
            print(f"  ✅ Conversion rates within normal range")

def analyze_match_distribution():
    """Analyze match status distribution."""
    print("\n📊 MATCH STATUS DISTRIBUTION")
    print("-" * 60)

    df = pd.read_sql("""
        SELECT status, COUNT(*) as cnt FROM matches GROUP BY status ORDER BY cnt DESC
    """, engine)

    if df.empty:
        print("  No matches found")
        return

    total = df['cnt'].sum()
    for _, row in df.iterrows():
        pct = row['cnt'] / total * 100
        bar = "█" * int(pct / 2)
        print(f"  {row['status']:<12} {row['cnt']:>8,} ({pct:>5.1f}%) {bar}")

    # Calculate confirmation rate
    confirmed = df[df['status'] == 'CONFIRMED']['cnt'].sum() if 'CONFIRMED' in df['status'].values else 0
    rejected = df[df['status'] == 'REJECTED']['cnt'].sum() if 'REJECTED' in df['status'].values else 0
    decided = confirmed + rejected

    print(f"\n  📈 Match Quality Metrics:")
    if decided > 0:
        print(f"      Confirmation rate: {confirmed/decided*100:.1f}%")
        print(f"      Rejection rate:    {rejected/decided*100:.1f}%")

def analyze_top_medications():

def analyze_activity_timeline():
```

## Phase 6: AI Parsing Quality Assessment

### Objective

Evaluate AI extraction accuracy and identify parsing issues.

### Expected Deliverables

- ✅ AI parsing success rate
- ✅ Unmapped medication analysis
- ✅ Consistency check for medication translations
- ✅ Review queue analysis

### Script: `06_ai_quality.py`

```python
#!/usr/bin/env python3
"""
Phase 6: AI Parsing Quality Assessment
========================================
Evaluates AI extraction accuracy and identifies parsing issues.
"""

import pandas as pd
from sqlalchemy import create_engine
import os

DATABASE_URL = os.getenv('PB_DATABASE_DSN', 'postgresql://postgres:postgres@localhost:5432/pharmabroker')
engine = create_engine(DATABASE_URL)

def analyze_parsing_success():
    """Calculate overall AI parsing success rate."""
    print("\n📊 AI PARSING SUCCESS METRICS")
    print("-" * 60)

    # Total processed messages
    total = pd.read_sql("""
        SELECT COUNT(*) FROM raw_messages WHERE processed_at IS NOT NULL
    """, engine).iloc[0, 0]

    # Successfully parsed (produced offers or requests)
    success = pd.read_sql("""
        SELECT COUNT(DISTINCT raw_message_id) FROM (
            SELECT raw_message_id FROM offers WHERE raw_message_id IS NOT NULL
            UNION
            SELECT raw_message_id FROM requests WHERE raw_message_id IS NOT NULL
        ) parsed
    """, engine).iloc[0, 0]

    # Failed messages
    failed = pd.read_sql("SELECT COUNT(*) FROM failed_messages", engine).iloc[0, 0]

    # Pending review
    pending_review = pd.read_sql("""
        SELECT COUNT(*) FROM review_queue WHERE status = 'PENDING'
    """, engine).iloc[0, 0]

    print(f"  Total processed:         {total:,}")
    print(f"  Successfully parsed:     {success:,} ({success/total*100:.1f}%)" if total else "  N/A")
    print(f"  Failed:                  {failed:,} ({failed/total*100:.1f}%)" if total else "")
    print(f"  Pending review:          {pending_review:,}")

    # Quality assessment
    if total > 0:
        success_rate = success / total * 100
        print(f"\n  📈 Quality Assessment:")
        if success_rate >= 70:
            print(f"      ✅ Good parsing rate: {success_rate:.1f}%")
        elif success_rate >= 50:
            print(f"      ⚠️  Moderate parsing rate: {success_rate:.1f}%")
        else:
            print(f"      ❌ Low parsing rate: {success_rate:.1f}% - investigate AI model")

def analyze_unmapped_medications():
    """Analyze medications that couldn't be mapped."""
    print("\n📊 UNMAPPED MEDICATIONS (TOP 20)")
    print("-" * 60)

    df = pd.read_sql("""
        SELECT raw_text, ai_output, count, reviewed, source_group
        FROM unmapped_medications
        ORDER BY count DESC
        LIMIT 20
    """, engine)

    if df.empty:
        print("  ✅ No unmapped medications found")
        return

    total_unmapped = pd.read_sql("SELECT COUNT(*) FROM unmapped_medications", engine).iloc[0, 0]
    reviewed_count = pd.read_sql("SELECT COUNT(*) FROM unmapped_medications WHERE reviewed = true", engine).iloc[0, 0]

    print(f"  Total unmapped: {total_unmapped:,}")
    print(f"  Reviewed: {reviewed_count:,} ({reviewed_count/total_unmapped*100:.1f}%)" if total_unmapped else "")
    print()

    for _, row in df.iterrows():
        reviewed = "✓" if row['reviewed'] else " "
        raw = str(row['raw_text'])[:25] if row['raw_text'] else 'N/A'
        ai = str(row['ai_output'])[:20] if row['ai_output'] else 'N/A'
        print(f"  [{reviewed}] {raw:<25} → {ai:<20} ({row['count']}x)")

def analyze_medication_consistency():
    """Check if same raw medication maps to multiple English names."""
    print("\n📊 MEDICATION MAPPING CONSISTENCY")
    print("-" * 60)

    df = pd.read_sql("""
        SELECT medication_raw,
               COUNT(DISTINCT medication) as unique_translations,
               ARRAY_AGG(DISTINCT medication) as translations,
               COUNT(*) as occurrences
        FROM offers
        WHERE medication_raw IS NOT NULL AND medication_raw != ''
        GROUP BY medication_raw
        HAVING COUNT(DISTINCT medication) > 1
        ORDER BY COUNT(DISTINCT medication) DESC
        LIMIT 10
    """, engine)

    if df.empty:
        print("  ✅ No inconsistent mappings found")
        return

    print(f"  ⚠️  Found {len(df)} medications with inconsistent translations:")
    for _, row in df.iterrows():
        print(f"    '{row['medication_raw'][:30]}' → {row['unique_translations']} translations")
        print(f"        {row['translations'][:3]}")  # Show first 3

def analyze_review_queue():
    """Analyze messages in review queue."""
    print("\n📊 REVIEW QUEUE ANALYSIS")
    print("-" * 60)

    df = pd.read_sql("""
        SELECT status, parse_pass, COUNT(*) as cnt, AVG(avg_confidence) as avg_conf
        FROM review_queue
        GROUP BY status, parse_pass
        ORDER BY parse_pass, status
    """, engine)

    if df.empty:
        print("  ✅ Review queue is empty")
        return

    print(f"  {'Status':<12} {'Pass':<6} {'Count':<8} {'Avg Confidence'}")
    print(f"  {'-'*12} {'-'*6} {'-'*8} {'-'*15}")
    for _, row in df.iterrows():
        print(f"  {row['status']:<12} {row['parse_pass']:<6} {row['cnt']:<8} {row['avg_conf']:.3f}" if row['avg_conf'] else f"  {row['status']:<12} {row['parse_pass']:<6} {row['cnt']:<8} N/A")

def main():
    print("=" * 70)
    print("PHASE 6: AI PARSING QUALITY ASSESSMENT")
    print("=" * 70)

    analyze_parsing_success()
    analyze_unmapped_medications()
    analyze_medication_consistency()
    analyze_review_queue()

    print("\n" + "=" * 70)
    print("AI QUALITY ASSESSMENT COMPLETE")
    print("=" * 70)

if __name__ == "__main__":
    main()
```
